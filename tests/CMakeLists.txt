# Each test is built from the Build: line in its source; ctest runs those
# that take no files
set(ODIN_TESTS_BY_HAND crypt_test)

file(GLOB TEST_FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*/*.cpp)
foreach(file ${TEST_FILES})
	file(READ ${file} text)
	if(NOT text MATCHES "Build: g\\+\\+ ([^\n\\\\]*(\\\\\n[^\n\\\\]*)*)")
		continue()
	endif()
	string(REGEX REPLACE "\\\\\n[ \t]*\\*" " " line "${CMAKE_MATCH_1}")
	string(REGEX REPLACE "(\\$\\(|`)pkg-config --(cflags|libs) sdl3(\\)|`)" "@sdl_\\2" line "${line}")
	separate_arguments(words UNIX_COMMAND "${line}")
	set(sources)
	set(flags)
	set(compile_flags)
	set(libs)
	set(name)
	set(next)
	foreach(word ${words})
		if(next STREQUAL "name")
			set(name ${word})
		elseif(next STREQUAL "include")
		elseif(next STREQUAL "isystem")
			list(APPEND compile_flags "SHELL:-isystem ${CMAKE_SOURCE_DIR}/${word}")
		elseif(word STREQUAL "-o")
			set(next name)
			continue()
		elseif(word STREQUAL "-I")
			set(next include)
			continue()
		elseif(word STREQUAL "-isystem")
			set(next isystem)
			continue()
		elseif(word MATCHES "\\.cpp$")
			list(APPEND sources ${CMAKE_SOURCE_DIR}/${word})
		elseif(word MATCHES "^src/(miniz|lzma)/.*\\.o$")
			list(APPEND sources $<TARGET_OBJECTS:odin_archive>)
		elseif(word STREQUAL "@sdl_cflags")
			list(APPEND flags ${SDL_CFLAGS})
		elseif(word STREQUAL "@sdl_libs")
			list(APPEND libs ${SDL_LIBS})
		elseif(word MATCHES "^-l")
			list(APPEND libs ${word})
		elseif(word MATCHES "^-")
			list(APPEND flags ${word})
		endif()
		set(next)
	endforeach()
	if(NOT name)
		message(FATAL_ERROR "${file}: no -o <name> on its Build: line")
	endif()
	list(REMOVE_DUPLICATES sources)
	add_executable(${name} ${sources})
	set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
	target_include_directories(${name} PRIVATE ${CMAKE_SOURCE_DIR}/src)
	target_compile_options(${name} PRIVATE ${flags} ${compile_flags})
	target_link_options(${name} PRIVATE ${flags})
	target_link_libraries(${name} PRIVATE ${libs})

	set(run ON)
	if(text MATCHES "\n \\* (Use|Run): +${name}([^\n.]*)")
		string(REGEX REPLACE "\\[[^]]*\\]" "" arguments "${CMAKE_MATCH_2}")
		if(arguments MATCHES "[^ \t]")
			set(run OFF)
		endif()
	endif()
	if(name IN_LIST ODIN_TESTS_BY_HAND)
		set(run OFF)
	endif()
	set(arguments)
	if(text MATCHES "in the folder given as its argument")
		set(arguments ${CMAKE_CURRENT_BINARY_DIR})
	elseif(text MATCHES "optional argument names a scratch file")
		set(arguments ${CMAKE_CURRENT_BINARY_DIR}/${name}.scratch)
	endif()
	if(run)
		add_test(NAME ${name} COMMAND ${name} ${arguments} WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
	endif()
endforeach()
